volumeAverage = ta.sma(volume, 20)

swingLow = ta.lowest(low, swingBars)
swingHigh = ta.highest(high, swingBars)
[htfClose, htfEMA] = request.security(syminfo.tickerid, "15", [close[1], ta.ema(close, 20)[1]], lookahead=barmerge.lookahead_on)
htfBull = not na(htfClose) and htfClose > htfEMA
htfBear = not na(htfClose) and htfClose < htfEMA
crossUp = ta.crossover(emaFast, emaSlow)
crossDown = ta.crossunder(emaFast, emaSlow)
pullbackUp = emaFast > emaSlow and low[1] <= emaFast[1] and close[1] <= emaFast[1] and close > emaFast and close > open
pullbackDown = emaFast < emaSlow and high[1] >= emaFast[1] and close[1] >= emaFast[1] and close < emaFast and close < open
longPattern = entryMode == "EMA Cross" ? crossUp : entryMode == "Trend Pullback" ? pullbackUp : crossUp or pullbackUp
shortPattern = entryMode == "EMA Cross" ? crossDown : entryMode == "Trend Pullback" ? pullbackDown : crossDown or pullbackDown
techLong = close > emaSlow and plusDI > minusDI and adxValue >= adxMin and rvol >= volMult and rsiValue > 50 and rsiValue < rsiUpper and (not useVWAP or close > vwapLine) and (not useHTF or htfBull)
techShort = close < emaSlow and minusDI > plusDI and adxValue >= adxMin and rvol >= volMult and rsiValue < 50 and rsiValue > rsiLower and (not useVWAP or close < vwapLine) and (not useHTF or htfBear)
macroValid = time_close >= macroFrom and time_close <= macroUntil
macroLong = not useMacroGate or (macroValid and (macroBias == "Neutral" or macroBias == "Bullish"))
macroShort = not useMacroGate or (macroValid and (macroBias == "Neutral" or macroBias == "Bearish"))
eventBlocked = useEventGate and time < eventUntil and time_close > eventFrom

f_price(float p) =>
    str.tostring(p, format.mintick)

type TradeFrame
    int side
    int startBar
    int closedAtEntry
    float entry
    float stop
    float tp1
    float tp2
    bool firstExitDone
    box reward
    box risk
    line entryLine
    line stopLine
    line tp1Line
    line tp2Line
    label entryLabel
    label stopLabel
    label tp1Label
    label tp2Label
    string status

var array<TradeFrame> history = array.new<TradeFrame>()
var TradeFrame active = na
var string lastReason = "WAIT CONFIRMED PATTERN"

var array<label> rejectedLabels = array.new<label>()

// Mark actual strategy exits, not high/low guesses about execution order.
if barstate.isconfirmed and strategy.closedtrades > observedClosed
    for i = observedClosed to strategy.closedtrades - 1
        if not na(active) and strategy.closedtrades.entry_bar_index(i) >= active.startBar
            if strategy.closedtrades.exit_id(i) == "TP1 / SL"
                active.firstExitDone := true
    observedClosed := strategy.closedtrades

if barstate.isconfirmed and not na(active)
    bool closed = bar_index > active.startBar and strategy.position_size == 0 and strategy.closedtrades > active.closedAtEntry
    if closed
        active.status := "CLOSED (TESTER)"
        int finish = math.max(active.startBar + 3, bar_index)
        int middle = int(math.avg(active.startBar, finish))
        box.set_right(active.reward, finish)
        box.set_right(active.risk, finish)
        line.set_x2(active.entryLine, finish)
        line.set_x2(active.stopLine, finish)
        line.set_x2(active.tp1Line, finish)
        line.set_x2(active.tp2Line, finish)
        label.set_x(active.entryLabel, middle)
        label.set_x(active.stopLabel, middle)
        label.set_x(active.tp1Label, finish)
        label.set_x(active.tp2Label, middle)
        label.set_text(active.entryLabel, (active.side == 1 ? "CALL" : "PUT") + " SIGNAL ENTRY " + f_price(active.entry) + "\nCLOSED — UNDERLYING TEST")
        label.set_color(active.entryLabel, color.gray)
        active := na

if barstate.isconfirmed
    if (cutoff or not inTest) and strategy.position_size != 0
        strategy.close_all(comment=not inTest ? "TEST WINDOW END" : "TIME CUTOFF", immediately=true, disable_alert=true)
    if longPattern or shortPattern
        int side = longPattern ? 1 : -1
        bool positionOK = na(active) or (reverseOnCross and active.side != side and (side == 1 ? crossUp : crossDown))
        bool technicalOK = not useTechnicalGate or (side == 1 ? techLong : techShort)
        bool macroOK = side == 1 ? macroLong : macroShort
        string techReason = adxValue < adxMin ? "ADX BELOW THRESHOLD" : na(rvol) or rvol < volMult ? "RELATIVE VOLUME LOW / MISSING" : useVWAP and (side == 1 ? close <= vwapLine : close >= vwapLine) ? "VWAP NOT ALIGNED" : useHTF and (side == 1 ? not htfBull : not htfBear) ? "CLOSED 15m NOT ALIGNED" : "RSI / DMI / TREND NOT ALIGNED"
        bool sizeOK = shareUnits * close <= strategy.equity * 0.95
        bool ready = not na(atrValue) and not na(swingLow) and not na(swingHigh)
        lastReason := not inTest ? "OUTSIDE TEST WINDOW" : not sizeOK ? "INSUFFICIENT MODEL EQUITY" : not ready ? "ATR / SWING WARMUP" : not inEntry ? "OUTSIDE ENTRY WINDOW" : cutoff ? "TIME CUTOFF" : eventBlocked ? "MANUAL EVENT BLACKOUT" : not macroOK ? (not macroValid ? "MACRO BIAS EXPIRED / MISSING" : "MANUAL MACRO BIAS") : not positionOK ? "ALREADY TRACKING SAME SIDE" : dayTrades >= maxTrades ? "DAILY ENTRY LIMIT" : not technicalOK ? techReason : "ENTRY ACCEPTED"
        if lastReason == "ENTRY ACCEPTED"
            // Opposite entries reverse the underlying Tester position; old drawings freeze.
            if not na(active)
                active.status := "OPPOSITE CROSS"
                label.set_text(active.entryLabel, (active.side == 1 ? "CALL" : "PUT") + " SIGNAL ENTRY " + f_price(active.entry) + "\nOPPOSITE CROSS")
                strategy.cancel("TP1 / SL")
                strategy.cancel("TP2 / SL")
            float entry = math.round_to_mintick(close)
            float stop = math.round_to_mintick(side == 1 ? math.min(swingLow - atrValue * stopBufferATR, entry - atrValue * atrMult) : math.max(swingHigh + atrValue * stopBufferATR, entry + atrValue * atrMult))
            float riskDistance = math.abs(entry - stop)
            float tp1 = math.round_to_mintick(entry + side * riskDistance * rr1)
            float tp2 = math.round_to_mintick(entry + side * riskDistance * rr2)
            int right = bar_index + frameBars
            string direction = side == 1 ? "CALL" : "PUT"
            string reference = "ATM REF " + str.tostring(math.round(entry), "#") + (side == 1 ? "C" : "P") + " x " + str.tostring(referenceContracts)
            reward = box.new(bar_index, math.max(entry, tp2), right, math.min(entry, tp2), border_color=color.teal, border_width=2, bgcolor=color.new(color.teal, 80), text="", text_color=color.teal)
            danger = box.new(bar_index, math.max(entry, stop), right, math.min(entry, stop), border_color=color.red, border_width=2, bgcolor=color.new(color.red, 82), text="", text_color=color.red)
            el = line.new(bar_index, entry, right, entry, color=color.blue, width=3)
            sl = line.new(bar_index, stop, right, stop, color=color.red, width=2)
            t1 = line.new(bar_index, tp1, right, tp1, color=color.teal, style=line.style_dashed)
            t2 = line.new(bar_index, tp2, right, tp2, color=color.teal, width=2)
            int middle = int(math.avg(bar_index, right))
            le = label.new(middle, entry, direction + " SIGNAL ENTRY " + f_price(entry) + "\nRisk/reward: " + str.tostring(rr2, "#.##") + " | UNDERLYING TEST", style=label.style_label_center, color=color.teal, textcolor=color.white, size=size.normal)
            ls = label.new(middle, stop, "Stop " + f_price(stop) + " | " + f_price(riskDistance) + " pts", style=side == 1 ? label.style_label_up : label.style_label_down, color=color.red, textcolor=color.white, size=size.normal)
            l1 = label.new(right, tp1, "TP1 " + f_price(tp1), style=label.style_label_left, color=color.teal, textcolor=color.white)
            l2 = label.new(middle, tp2, "Target " + f_price(tp2) + " | " + f_price(math.abs(tp2 - entry)) + " pts", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.teal, textcolor=color.white)
            active := TradeFrame.new(side, bar_index, strategy.closedtrades, entry, stop, tp1, tp2, false, reward, danger, el, sl, t1, t2, le, ls, l1, l2, "OPEN")
            array.push(history, active)
            dayTrades += 1
            longSignal := side == 1
            shortSignal := side == -1
            strategy.entry(side == 1 ? "CALL" : "PUT", side == 1 ? strategy.long : strategy.short, qty=shareUnits, disable_alert=true)
        else if showRejected
            rejection = label.new(bar_index, side == 1 ? low : high, "NO ENTRY\n" + lastReason, style=side == 1 ? label.style_label_up : label.style_label_down, color=color.gray, textcolor=color.white, size=size.tiny)
            array.push(rejectedLabels, rejection)
            while array.size(rejectedLabels) > 30
                label.delete(array.shift(rejectedLabels))
